7. **PDF 下载不在普通 API 函数里走拦截器解包** —— 后端返回裸二进制，必须：
   ```ts
   /** GET /api/salary/payslips/:id/payslip/pdf · salary:payslip:generate · 二进制下载 */
   export async function downloadPayslipPdf(id: string, fallbackName = 'payslip.pdf'): Promise<void> {
     const blob = await http.get(PAYSLIP_PATHS.pdf(id), { responseType: 'blob' }) as unknown as Blob;
     const url = URL.createObjectURL(blob);
     const a = document.createElement('a');
     a.href = url;
     a.download = fallbackName;
     a.click();
     URL.revokeObjectURL(url);
   }
   ```
   （http 拦截器返回 `response.data`，`responseType:'blob'` 时即 Blob；文件名可从后端 Content-Disposition 取，若取不到用 `工资条-{period}.pdf` 兜底）
