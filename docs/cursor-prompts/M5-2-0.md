### 3.4 错误码段位（M5-2-0 0 新错误码）

**M5-2-0 不分配任何错误码**（前端不涉及业务错误码定义，仅消费后端错误码）。

**前端错误处理**：
- 后端返回 `{ code: 7xxxx, message: 'xxx' }` 时，前端根据 `code` 显示对应 toast
- 401 触发 token 刷新 / 重定向登录
- 403 触发权限不足 toast
- 500 触发服务器错误 toast
- 网络错误触发"网络异常" toast

**`api/types.ts` 定义**：
```ts
export type ErrorCode = number  // 5 位数字错误码
export interface ApiError {
  code: ErrorCode
  message: string
  details?: Record<string, unknown>
}

import axios, { AxiosInstance, InternalAxiosRequestConfig, AxiosResponse } from 'axios'
import { ElMessage } from 'element-plus'
import { getToken, setToken, clearToken, getRefreshToken } from '@/utils/auth'
import { refreshToken } from './auth'

const http: AxiosInstance = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL || '/api',
  timeout: 30000,
  headers: { 'Content-Type': 'application/json' },
})

// 请求拦截器：自动加 token
http.interceptors.request.use(
  (config: InternalAxiosRequestConfig) => {
    const token = getToken()
    if (token && config.headers) {
      config.headers.Authorization = `Bearer ${token}`
    }
    return config
  },
  (error) => Promise.reject(error),
)

// 响应拦截器：401 自动刷新 + 错误 toast
http.interceptors.response.use(
  (response: AxiosResponse) => response.data,
  async (error) => {
    const originalRequest = error.config
    if (error.response?.status === 401 && !originalRequest._retry) {
      originalRequest._retry = true
      try {
        const refreshTokenValue = getRefreshToken()
        if (refreshTokenValue) {
          const res = await refreshToken(refreshTokenValue)
          setToken(res.accessToken, res.refreshToken)
          originalRequest.headers.Authorization = `Bearer ${res.accessToken}`
          return http(originalRequest)
        }
      } catch (refreshError) {
        clearToken()
        window.location.href = '/login'
        return Promise.reject(refreshError)
      }
    }
    const apiError = error.response?.data as ApiError
    if (apiError?.code) {
      ElMessage.error(apiError.message || '请求失败')
    } else {
      ElMessage.error('网络异常')
    }
    return Promise.reject(error)
  },
)

import { defineStore } from 'pinia'
import { ref } from 'vue'

const CACHE_TTL = 5 * 60 * 1000  // 5 分钟

export const useDictStore = defineStore('dict', () => {
  const cache = ref<Map<string, { value: unknown; expireAt: number }>>(new Map())

  function get<T>(key: string, fetcher: () => Promise<T>): Promise<T> {
    const cached = cache.value.get(key)
    if (cached && cached.expireAt > Date.now()) {
      return Promise.resolve(cached.value as T)
    }
    return fetcher().then((value) => {
      cache.value.set(key, { value, expireAt: Date.now() + CACHE_TTL })
      return value
    })
  }

  function clear() {
    cache.value.clear()
  }

  return { get, clear }
})

    _retry?: boolean

}

  baseURL: import.meta.env.VITE_API_BASE_URL || '/api',
  timeout: 30000,
  headers: { 'Content-Type': 'application/json' },

      config.headers.Authorization = `Bearer ${token}`
    }

          const res = await apiRefreshToken(refreshTokenValue)
          setToken(res.accessToken, res.refreshToken)
          originalRequest.headers.Authorization = `Bearer ${res.accessToken}`
          return http(originalRequest)
        }
      } catch (refreshError) {
        clearToken()
        window.location.href = '/login'
        return Promise.reject(refreshError)
      }
    }
    const apiError = error.response?.data as ApiError
    if (apiError?.code) {
      ElMessage.error(apiError.message || '请求失败')
    } else {
      ElMessage.error('网络异常')
    }
    return Promise.reject(error)
  },
)

export default http
```
