```prisma
// ==================== M2-B5 出差 ====================
// 状态机：draft → submitted → approved / rejected / cancelled
// 关联 employee（A2 已有）+ approvalInstance（M0.5-1 已有）
// 差旅补助（allowance_amount）由 service 函数计算，写入字段；M4 薪酬切片读取聚合
// 出差期间 GPS 打卡由 B2 attendance_records 独立处理（B5 不联动）
model BusinessTrip {
  id            String   @id @default(uuid()) @db.Uuid
  // 关联员工
  employeeId    String   @map("employee_id") @db.Uuid
  companyId     String   @map("company_id") @db.Uuid
  departmentId  String   @map("department_id") @db.Uuid
  // 出差基本信息
  destination   String   @db.VarChar(100)  // 出差城市，如 '北京' / '上海' / '深圳'
  // 出差时间
  startDate     DateTime @map("start_date") @db.Date
  endDate       DateTime @map("end_date") @db.Date
  // 出差天数（service 层 calculateTripDays 计算，**默认排除周末** configs.trip.weekend_inclusive）
  totalDays     Decimal  @map("total_days") @db.Decimal(4, 1)
  // 出差事由
  reason        String   @db.Text
  // 关联项目（**B5 简化**为字符串，B5 不创建 projects 表；如需关联可二期建 projects 表）
  projectCode   String?  @map("project_code") @db.VarChar(50)  // 项目代码
  // 差旅补助（service 层 calculateTravelAllowance 计算）
  allowanceAmount Decimal? @map("allowance_amount") @db.Decimal(10, 2)
  // 城市 tier（configs.trip.city_tier_mapping 查）
  cityTier      String?  @map("city_tier") @db.VarChar(20)  // tier1 / tier2 / tier3
  // 职级系数（configs.trip.level_tier_rates 查）
  levelRate     Decimal? @map("level_rate") @db.Decimal(3, 2)  // 1.0 / 1.2 / 1.5
  // 状态机
  status        String   @default("draft") @db.VarChar(20)
  // draft / submitted / approved / rejected / cancelled
  // 审批流（M0.5-1）
  approvalInstanceId String? @unique @map("approval_instance_id") @db.Uuid
  // 审批结果
  approvedAt    DateTime? @map("approved_at") @db.Timestamptz(6)
  approvedBy    String?  @map("approved_by") @db.Uuid
  rejectedAt    DateTime? @map("rejected_at") @db.Timestamptz(6)
  rejectedBy    String?  @map("rejected_by") @db.Uuid
  rejectedReason String? @map("rejected_reason") @db.Text
  // 取消
  cancelledAt   DateTime? @map("cancelled_at") @db.Timestamptz(6)
  cancelledBy   String?  @map("cancelled_by") @db.Uuid
  cancelledReason String? @map("cancelled_reason") @db.Text
  // 元数据
  createdBy     String?  @map("created_by") @db.Uuid
  createdAt     DateTime @default(now()) @map("created_at") @db.Timestamptz(6)
  updatedAt     DateTime @updatedAt @map("updated_at") @db.Timestamptz(6)

  @@index([employeeId, status])
  @@index([companyId, status, startDate])
  @@index([departmentId, status, startDate])
  @@index([status, startDate])
  @@index([destination])
  @@map("business_trips")
}
```
