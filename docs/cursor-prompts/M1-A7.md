【A7 业务背景（V1.2 §二.2.4）】
- 合同模板库：5 类（劳动合同 / 实习协议 / 顾问协议 / 劳务合同 / 保密协议）
- 合同状态机：draft → pending_signature → signing → signed / expired / cancelled
  - draft：HR 起草中
  - pending_signature：草稿完成待发起电子签
  - signing：已发起电子签，等待签署方完成
  - signed：双方/多方签署完成，合同生效
  - expired：合同到期未续签（自动流转）
  - cancelled：合同被作废
- 合同附件：扫描件 PDF/JPG，单文件 ≤10MB（**A7 简化：只接 attachmentUrl 字符串**）
- 到期预警：30/15/7 天三级（V1.2 §二.2.4）
- 电子签：一期 mock（不接 e-签宝真实 SaaS）

【A7 状态机（详细）】
```
draft ──update──→ draft（仅草稿可改）
  │                
  ├──cancel──┐
  ↓         ↓
draft → pending_signature ──sign──→ signing ──callback──→ signed
                                   │                        │
                                   ├──cancel──┐             ├──expire──→ expired
                                   ↓          ↓             │
                                cancelled cancelled         expired
```

// ==================== M1-A7 合同管理 ====================
// 状态机：draft → pending_signature → signing → signed / expired / cancelled
// 关联 employee（A2 已有）+ approval_instance（M0.5-1 已有）
// 电子签：e-签宝 mock（configs.contract.esign_provider='mock'）
model ContractRecord {
  id            String   @id @default(uuid()) @db.Uuid
  // 关联员工
  employeeId    String   @map("employee_id") @db.Uuid
  // 合同基本信息
  contractNo    String   @unique @map("contract_no") @db.VarChar(50)  // 合同编号（自动生成）
  contractType  String   @map("contract_type") @db.VarChar(50)
  // formal / intern / consultant / labor / nda
  title         String   @db.VarChar(200)  // 合同标题
  // 合同期限
  startDate     DateTime @map("start_date") @db.Date
  endDate       DateTime @map("end_date") @db.Date
  // 试用期（A4 转正流程用，A7 同步）
  probationMonths Int?  @map("probation_months")
  // 薪资（草稿可填，签约时锁定）
  baseSalary    Decimal? @map("base_salary") @db.Decimal(10, 2)
  // 工作内容
  position      String?  @db.VarChar(100)
  workLocation  String?  @map("work_location") @db.VarChar(200)
  // 模板（configs.contract.templates 路径）
  templateKey   String   @map("template_key") @db.VarChar(50)
  // 附件（JSON 数组，最多 5 个，A7 简化只接 URL 字符串）
  attachments   Json?    // [{ name, url, type, size, uploadedAt }]
  // 签署方（JSON 数组，e-签宝 mock 模式硬编码员工 + HR）
  signatories   Json?    // [{ name, role, phone, email, signed }]
  // 电子签（mock）
  esignProvider String  @default("mock") @map("esign_provider") @db.VarChar(20)
  esignFlowId   String? @map("esign_flow_id") @db.VarChar(100)  // e-签宝 flowId（mock 模式生成 mock-id）
  esignFlowUrl  String? @map("esign_flow_url") @db.Text  // e-签宝签署链接（mock 模式为空）
  // 审批流
  approvalInstanceId String? @unique @map("approval_instance_id") @db.Uuid
  // 状态机
  status        String   @default("draft") @db.VarChar(20)
  // draft / pending_signature / signing / signed / expired / cancelled
  // 签约时间
  signedAt      DateTime? @map("signed_at") @db.Timestamptz(6)
  signedBy      String?  @map("signed_by") @db.Uuid  // 最后签约人 userId
  // 到期预警调度（V1.2 §二.2.4 三级预警）
  expireWarned30d Boolean @default(false) @map("expire_warned_30d")
  expireWarned15d Boolean @default(false) @map("expire_warned_15d")
  expireWarned7d  Boolean @default(false) @map("expire_warned_7d")
  // 流转到 expired 的时间
  expiredAt     DateTime? @map("expired_at") @db.Timestamptz(6)
  // 取消
  cancelledAt   DateTime? @map("cancelled_at") @db.Timestamptz(6)
  cancelledBy   String?  @map("cancelled_by") @db.Uuid
  cancelledReason String? @map("cancelled_reason") @db.Text
  // 元数据
  createdBy     String?  @map("created_by") @db.Uuid
  createdAt     DateTime @default(now()) @map("created_at") @db.Timestamptz(6)
  updatedAt     DateTime @updatedAt @map("updated_at") @db.Timestamptz(6)

  @@index([employeeId, status])
  @@index([status, endDate])  // 到期预警 BullMQ 扫描
  @@index([contractType, status])
  @@index([createdBy, status])
  @@map("contract_records")
}
```

**注意**：

- `@@index([contractType, status])` —— 按类型统计
- `esignFlowId / esignFlowUrl` —— mock 模式下 esignFlowId = 'mock-' + uuid，esignFlowUrl = null
- **不修改 A1-A6 任何已有表**（包括 employees / companies / departments / 等）
