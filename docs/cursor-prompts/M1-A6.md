### 任务 1: 更新 Prisma schema（+ 2 张表 + 索引）

**新增模型**（**完整定义**，不要"列关键字段"）：

```prisma
// ==================== M1-A6 离职流程 ====================
// 状态机：draft → handover_pending → submitted → approved
//                      → certificate_issued / rejected / cancelled
// 关联 employee（A2 已有）+ approval_instance（M0.5-1 已有）+ user（M0 已有）
model OffboardingRecord {
  id            String   @id @default(uuid()) @db.Uuid
  // 关联员工（必须 active 状态）
  employeeId    String   @map("employee_id") @db.Uuid
  // 离职类型
  resignationType String  @map("resignation_type") @db.VarChar(20)
  // employee_initiated / company_initiated
  // 离职原因
  reason        String?  @db.Text
  // 离职生效日（最后工作日）
  lastWorkingDate DateTime @map("last_working_date") @db.Date
  // 工作交接状态
  handoverCompleted Boolean @default(false) @map("handover_completed")
  handoverCompletedBy String? @map("handover_completed_by") @db.Uuid
  handoverCompletedAt DateTime? @map("handover_completed_at") @db.Timestamptz(6)
  // 审批流（M0.5-1）
  approvalInstanceId String? @unique @map("approval_instance_id") @db.Uuid
  // 离职证明
  certificateIssued Boolean @default(false) @map("certificate_issued")
  certificateIssuedAt DateTime? @map("certificate_issued_at") @db.Timestamptz(6)
  certificateIssuedBy String? @map("certificate_issued_by") @db.Uuid
  certificateNumber String? @unique @map("certificate_number") @db.VarChar(50)
  certificateUrl   String?  @map("certificate_url") @db.Text
  // 账号禁用
  accountDisabled Boolean @default(false) @map("account_disabled")
  accountDisabledAt DateTime? @map("account_disabled_at") @db.Timestamptz(6)
  // 档案归档
  archivedAt DateTime? @map("archived_at") @db.Timestamptz(6)
  archiveRetentionYears Int @default(5) @map("archive_retention_years")
  // configs.archive.years 写入
  // 状态机
  status        String   @default("draft") @db.VarChar(20)
  // draft / handover_pending / submitted / approved
  // certificate_issued / rejected / cancelled
  // 元数据
  createdBy     String?  @map("created_by") @db.Uuid
  createdAt     DateTime @default(now()) @map("created_at") @db.Timestamptz(6)
  updatedAt     DateTime @updatedAt @map("updated_at") @db.Timestamptz(6)
  cancelledAt   DateTime? @map("cancelled_at") @db.Timestamptz(6)
  cancelledBy   String?  @map("cancelled_by") @db.Uuid
  cancelledReason String? @map("cancelled_reason") @db.Text

  tasks HandoverTask[]

  @@index([employeeId, status])
  @@index([status, lastWorkingDate])
  @@index([createdBy, status])
  @@map("offboarding_records")
}

// 工作交接任务（从 configs.offboarding.handover_template 模板生成）
model HandoverTask {
  id          String   @id @default(uuid()) @db.Uuid
  offboardingId String  @map("offboarding_id") @db.Uuid
  // 任务名（来自 handover_template）
  name        String   @db.VarChar(100)
  // '工作文档交接' / '客户/项目交接' / '财务/物资交接' / '系统账号交接' / '未了事项说明'
  // 任务类别（5 类对应 5 个模板项）
  category    String   @db.VarChar(50)
  // document / client / finance / account / misc
  // 接收方（接收工作的人）
  ownerId     String?  @map("owner_id") @db.Uuid
  // 状态
  status      String   @default("pending") @db.VarChar(20)
  // pending / in_progress / done / skipped
  // 备注
  remark      String?  @db.Text
  // 完成时间
  completedAt DateTime? @map("completed_at") @db.Timestamptz(6)
  completedBy String?  @map("completed_by") @db.Uuid
  createdAt   DateTime @default(now()) @map("created_at") @db.Timestamptz(6)
  updatedAt   DateTime @updatedAt @map("updated_at") @db.Timestamptz(6)

  offboarding OffboardingRecord @relation(fields: [offboardingId], references: [id], onDelete: Cascade)

  @@index([offboardingId, status])
  @@index([ownerId, status])
  @@map("handover_tasks")
}
```

**注意**：
- `@@unique([certificateNumber])` —— 证书编号唯一
- `@@index([status, lastWorkingDate])` —— BullMQ 定时任务扫描用
- `HandoverTask` 走外键 Cascade，删除 offboarding 自动删 tasks
- **不修改 A1-A4 任何已有表**（不加 Employee / User 反向 relation，避免改 A1-A4 model）

## 优化理念

| 方向 | 效果 |
|---|---|
| 详尽提示词（25-28KB） | 关键决策点全部锁死，Cursor 一次到位，token 反而少 |
| 简化提示词（15KB） | Cursor 推断/补全消耗大量 token，返工风险高，**反而多消耗** |
| 细化 = 优化 | 关键决策点（schema 完整定义 / 函数参数 + 关键调用 / 错误码触发条件 / 测试断言细节）**详尽**；样板内容（Zod/controller/routes 风格）**引用前切片**避免重复 |
