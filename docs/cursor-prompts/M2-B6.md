// ==================== M2-B6 月度考勤汇总 ====================
// 状态机：draft → employee_confirmed → hr_locked（不可跳级）
// 关联 employee（A2 已有）

model MonthlySummary {
  id            String   @id @default(uuid()) @db.Uuid
  // 关联员工
  employeeId    String   @map("employee_id") @db.Uuid
  companyId     String   @map("company_id") @db.Uuid
  departmentId  String   @map("department_id") @db.Uuid
  // 月份
  year          Int      @db.SmallInt
  month         Int      @db.SmallInt  // 1-12
  // 报表字段（聚合 B1-B5 数据）
  workDays      Decimal  @map("work_days") @db.Decimal(4, 1)  // 出勤天数
  lateCount     Int      @default(0) @map("late_count")
  earlyLeaveCount Int    @default(0) @map("early_leave_count")
  missingCount  Int      @default(0) @map("missing_count")
  leaveDays     Decimal  @default(0) @map("leave_days") @db.Decimal(5, 1)
  leaveHours    Decimal  @default(0) @map("leave_hours") @db.Decimal(6, 1)
  overtimeHours Decimal  @default(0) @map("overtime_hours") @db.Decimal(6, 1)
  tripDays      Decimal  @default(0) @map("trip_days") @db.Decimal(4, 1)
  // **B6 关键承诺**：调休余额
  compBalance   Decimal  @default(0) @map("comp_balance") @db.Decimal(6, 1)  // 调休余额（天数）
  // 状态机
  status        String   @default("draft") @db.VarChar(20)
  // draft / employee_confirmed / hr_locked
  // 员工确认
  employeeConfirmedAt DateTime? @map("employee_confirmed_at") @db.Timestamptz(6)
  employeeConfirmedBy String?  @map("employee_confirmed_by") @db.Uuid
  // HR 锁定
  hrLockedAt    DateTime? @map("hr_locked_at") @db.Timestamptz(6)
  hrLockedBy    String?  @map("hr_locked_by") @db.Uuid
  // 备注
  remark        String?  @db.Text
  // 元数据
  createdBy     String?  @map("created_by") @db.Uuid
  createdAt     DateTime @default(now()) @map("created_at") @db.Timestamptz(6)
  updatedAt     DateTime @updatedAt @map("updated_at") @db.Timestamptz(6)

  @@unique([employeeId, year, month])  // 同员工同月唯一
  @@index([companyId, year, month])
  @@index([departmentId, year, month])
  @@index([status])
  @@map("monthly_summaries")
}
