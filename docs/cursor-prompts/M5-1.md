## 4. 实施任务

### 4.1 子任务 5.1.1：Dockerfile.server（多阶段 build）

**1 新文件**（`Dockerfile.server` 项目根 + `.dockerignore` server/）

**多阶段 build 结构**（**4 阶段**）：

```dockerfile
# ===== Stage 1: deps（pnpm fetch）=====
FROM node:20-alpine AS deps
WORKDIR /app
RUN corepack enable && corepack prepare pnpm@8.15.0 --activate
COPY pnpm-workspace.yaml package.json pnpm-lock.yaml* ./
COPY server/package.json ./server/
COPY client/package.json ./client/
RUN pnpm install --frozen-lockfile --filter=hrms-server...

# ===== Stage 2: builder（prisma generate + tsc）=====
FROM node:20-alpine AS builder
WORKDIR /app
RUN corepack enable && corepack prepare pnpm@8.15.0 --activate
COPY --from=deps /app/node_modules ./node_modules
COPY --from=deps /app/server/node_modules ./server/node_modules
COPY . .
RUN cd server && npx prisma generate
RUN cd server && pnpm build

# ===== Stage 3: prod-deps（生产依赖）=====
FROM node:20-alpine AS prod-deps
WORKDIR /app
RUN corepack enable && corepack prepare pnpm@8.15.0 --activate
COPY pnpm-workspace.yaml package.json pnpm-lock.yaml* ./
COPY server/package.json ./server/
RUN pnpm install --prod --frozen-lockfile --filter=hrms-server...

# ===== Stage 4: runner（生产运行时，非 root 用户）=====
FROM node:20-alpine AS runner
WORKDIR /app
ENV NODE_ENV=production
ENV PORT=3001
# 创建非 root 用户
RUN addgroup -g 1001 -S nodejs && adduser -S hrms -u 1001
# 复制生产构建产物
COPY --from=builder --chown=hrms:nodejs /app/server/dist ./server/dist
```

**nginx-client.conf**（client 容器内 nginx 配置）：

```nginx
server {
  listen 80;
  server_name _;
  
  # 客户端请求体上限（上传文件 10MB）
  client_max_body_size 10M;
  
  # gzip 压缩
  gzip on;
  gzip_vary on;
  gzip_min_length 1024;
  gzip_types text/plain text/css text/xml text/javascript application/javascript application/json application/xml+rss image/svg+xml;
  
  # 安全 headers
  add_header X-Frame-Options "SAMEORIGIN" always;
  add_header X-Content-Type-Options "nosniff" always;
  add_header X-XSS-Protection "1; mode=block" always;
  add_header Referrer-Policy "no-referrer-when-downgrade" always;
  
  # 静态资源（Vite build 产物，含 hash 永久缓存）
  location /assets/ {
    root /usr/share/nginx/html;
    expires 1y;
    add_header Cache-Control "public, immutable";
    access_log off;
  }
  
  # 上传文件
  location /uploads/ {
    root /usr/share/nginx/html;
    expires 7d;
    add_header Cache-Control "public";
  }
  
  # API 反代（/api/* → server:3001）
  location /api/ {
    proxy_pass http://server:3001/api/;
    proxy_http_version 1.1;
    proxy_set_header Upgrade $http_upgrade;
    proxy_set_header Connection 'upgrade';
    proxy_set_header Host $host;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto $scheme;
    proxy_cache_bypass $http_upgrade;
    proxy_connect_timeout 60s;
    proxy_send_timeout 60s;
    proxy_read_timeout 60s;
  }
  
  # SPA fallback（所有非 /api/ 非 /assets/ 请求 → index.html）
  location / {
    root /usr/share/nginx/html;
    try_files $uri $uri/ /index.html;
    add_header Cache-Control "no-cache, no-store, must-revalidate";
  }
  
  # 健康检查（Docker healthcheck）
  location = /healthz {
    access_log off;
    return 200 "ok\n";
    add_header Content-Type text/plain;
  }
  
  # 404 / 500
  error_page 404 /index.html;
  error_page 500 502 503 504 /50x.html;
  location = /50x.html {
    root /usr/share/nginx/html;
  }
}
```
