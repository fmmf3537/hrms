  - 早退：< endTime - early_leave_threshold 打卡为早退
  - 缺卡：当日无任何 clock-in / clock-out 记录
  - 综合工时（生产/外场）：按月/季综合
  - 不定时工时（高管/销售）：不打卡

```prisma
// ==================== M2-B2 打卡管理 ====================
// 状态机：clock-in 直接 approved / manual 走 draft→submitted→approved/rejected / import 走 approved
// 关联 employee（A2 已有）+ shiftAssignment（M2-B1 已有）+ approvalInstance（M0.5-1 已有）
model AttendanceRecord {
  id            String   @id @default(uuid()) @db.Uuid
  // 关联员工
  employeeId    String   @map("employee_id") @db.Uuid
  companyId     String   @map("company_id") @db.Uuid
  departmentId  String   @map("department_id") @db.Uuid
  // 关联班次分配（M2-B1，可能为空：员工未分配班次）
  shiftAssignmentId String? @map("shift_assignment_id") @db.Uuid
  // 打卡时间
  clockInTime   DateTime? @map("clock_in_time") @db.Timestamptz(6)
  clockOutTime  DateTime? @map("clock_out_time") @db.Timestamptz(6)
  // 打卡类型
  clockType     String   @map("clock_type") @db.VarChar(20)
  // wifi / gps / manual / imported
  // 来源
  source        String   @map("source") @db.VarChar(20)
  // app / admin / import
  // WiFi 打卡字段
  wifiSsid      String?  @map("wifi_ssid") @db.VarChar(100)
  wifiMac       String?  @map("wifi_mac") @db.VarChar(50)
  // GPS 打卡字段
  gpsLat        Decimal? @map("gps_lat") @db.Decimal(10, 7)
  gpsLng        Decimal? @map("gps_lng") @db.Decimal(10, 7)
  gpsAccuracy   Decimal? @map("gps_accuracy") @db.Decimal(10, 2)  // 精度（米）
  gpsAddress    String?  @map("gps_address") @db.VarChar(500)  // mock 地址（B2 不接腾讯地图 API）
  // 导入数据来源
  importedExternalId String? @unique @map("imported_external_id") @db.VarChar(100)
  importedFrom String?  @map("imported_from") @db.VarChar(50)  // 'deli-e-plus-v1'
  importedAt   DateTime? @map("imported_at") @db.Timestamptz(6)
  // 异常判定（service 层计算）
  isLate        Boolean  @default(false) @map("is_late")
  lateMinutes   Int      @default(0) @map("late_minutes")
  isEarlyLeave  Boolean  @default(false) @map("is_early_leave")
  earlyLeaveMinutes Int @default(0) @map("early_leave_minutes")
  isMissing     Boolean  @default(false) @map("is_missing")
  // 状态机
  status        String   @default("approved") @db.VarChar(20)
  // approved / pending / rejected
  // 补卡关联
  isManual      Boolean  @default(false) @map("is_manual")
  manualReason  String?  @map("manual_reason") @db.Text
  approvalInstanceId String? @unique @map("approval_instance_id") @db.Uuid
  // 备注
  remark        String?  @db.Text
  // 元数据
  createdBy     String?  @map("created_by") @db.Uuid
  createdAt     DateTime @default(now()) @map("created_at") @db.Timestamptz(6)
  updatedAt     DateTime @updatedAt @map("updated_at") @db.Timestamptz(6)
  rejectedAt    DateTime? @map("rejected_at") @db.Timestamptz(6)
  rejectedBy    String?  @map("rejected_by") @db.Uuid
  rejectedReason String? @map("rejected_reason") @db.Text

  @@unique([employeeId, clockInTime])  // 同员工同 clock-in 时间唯一
  @@index([employeeId, clockInTime])
  @@index([companyId, clockInTime])
  @@index([departmentId, clockInTime])
  @@index([status, clockInTime])
  @@index([isManual, status])
  @@index([importedExternalId])
  @@map("attendance_records")
}
```
