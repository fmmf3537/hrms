**新增模型**（**完整定义**，不要"列关键字段"）：

```prisma
// ==================== M2-B1 班次定义与排班 ====================
// 状态机：draft → active → archived
// 关联 employee（A2 已有）+ department（A1+A2 已有）
//                    + audit（M0.5 已有）
model ShiftTemplate {
  id            String   @id @default(uuid()) @db.Uuid
  // 班次基本信息
  code          String   @unique @db.VarChar(50)  // 班次编码（公司内唯一）
  name          String   @db.VarChar(100)  // 班次名称（如"标准白班"）
  // 工时制类型
  shiftType     String   @map("shift_type") @db.VarChar(20)
  // standard / comprehensive / flexible
  // 上下班时间（HH:mm 格式字符串）
  startTime     String   @map("start_time") @db.VarChar(5)  // '09:00'
  endTime       String   @map("end_time") @db.VarChar(5)  // '18:00'
  // 午休
  breakStart    String?  @map("break_start") @db.VarChar(5)  // '12:00'
  breakEnd      String?  @map("break_end") @db.VarChar(5)  // '13:30'
  breakDuration Int?     @map("break_duration")  // 午休分钟（冗余）
  // 工作时长
  workHours     Decimal  @map("work_hours") @db.Decimal(4, 2)  // 8.00
  // 弹性时间（分钟）
  flexMinutes   Int      @default(30) @map("flex_minutes")
  // 有效日期
  effectiveFrom DateTime @map("effective_from") @db.Date
  effectiveTo   DateTime? @map("effective_to") @db.Date
  // 适用公司（多公司支持，A1 已有 Company）
  companyId     String   @map("company_id") @db.Uuid
  // 描述
  description   String?  @db.Text
  // 状态机
  status        String   @default("draft") @db.VarChar(20)
  // draft / active / archived
  // 元数据
  createdBy     String?  @map("created_by") @db.Uuid
  createdAt     DateTime @default(now()) @map("created_at") @db.Timestamptz(6)
  updatedAt     DateTime @updatedAt @map("updated_at") @db.Timestamptz(6)
  archivedAt    DateTime? @map("archived_at") @db.Timestamptz(6)
  archivedBy    String?  @map("archived_by") @db.Uuid

  assignments ShiftAssignment[]

  @@index([companyId, status])
  @@index([shiftType, status])
  @@index([effectiveFrom, effectiveTo])
  @@map("shift_templates")
}

// 班次分配记录（员工/部门级）
model ShiftAssignment {
  id            String   @id @default(uuid()) @db.Uuid
  // 关联班次
  shiftId       String   @map("shift_id") @db.Uuid
  // 分配对象类型
  assigneeType  String   @map("assignee_type") @db.VarChar(20)
  // employee / department
  // 关联员工/部门（互斥）
  employeeId    String?  @map("employee_id") @db.Uuid
  departmentId  String?  @map("department_id") @db.Uuid
  // 生效日期范围
  effectiveFrom DateTime @map("effective_from") @db.Date
  effectiveTo   DateTime? @map("effective_to") @db.Date
  // 分配备注
  remark        String?  @db.Text
  // 元数据
  createdBy     String?  @map("created_by") @db.Uuid
  createdAt     DateTime @default(now()) @map("created_at") @db.Timestamptz(6)
  updatedAt     DateTime @updatedAt @map("updated_at") @db.Timestamptz(6)

  shift ShiftTemplate @relation(fields: [shiftId], references: [id], onDelete: Restrict)

  @@index([shiftId, effectiveFrom])
  @@index([employeeId, effectiveFrom])
  @@index([departmentId, effectiveFrom])
  @@map("shift_assignments")
}
```

**注意**：
- `ShiftTemplate.code` unique（公司内唯一）
- `ShiftAssignment` 关联 employeeId / departmentId 互斥（约束：要么 employeeId 有值，要么 departmentId 有值）
- `@@index` 覆盖：company+status、type+status、date 范围、employee 维度、department 维度
- **不修改 M0.5 + M1 任何已有表**（包括 employees / companies / departments）
