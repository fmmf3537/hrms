**判定算法**（V1.2 §二.5.1）：
```ts
function calcGrade(finalScore: number, thresholds: { S, A, B, C, D }): PerformanceGrade {
  if (finalScore >= thresholds.S) return 'S';
  if (finalScore >= thresholds.A) return 'A';
  if (finalScore >= thresholds.B) return 'B';
  if (finalScore >= thresholds.C) return 'C';
  return 'D';
}
```

export async function enhanceUpdateCoefficients(

/**
 * 单条记录等级判定（基于 finalScore + thresholds）
 * @param actorId 操作人 ID（HR/admin）
 * @param recordId record ID
 * @param options { force?: boolean } - force=true 时允许覆盖已有 finalGrade
 * @returns { recordId, finalScore, oldGrade, newGrade, threshold }
 * @throws AppError(404) record 不存在
 * @throws AppError(400, 72601) record.finalScore 为 null
 * @throws AppError(400, 72603) finalScore 越界

 *  1. 查询 record → 不存在抛 404
 *  2. 校验 record.status === 'ceo_approved'（未审批不能判定等级）→ 否则抛 400（V1.2 §二.5.1）

 *  9. 返回 { recordId, finalScore, oldGrade, newGrade, threshold }
 *  注：force=true 覆盖 finalGrade 时，写 audit details 含 { oldGrade, newGrade, reason: 'manual_override' }
 */

  actorId: string,
  recordId: string,
  options?: { force?: boolean }
