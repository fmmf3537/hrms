// ==================== M1-A1 组织架构 ====================
model Company {
  id          String   @id @default(uuid()) @db.Uuid
  code        String   @db.VarChar(20) // 法人代码如 XACH / XACX / SCXH
  name        String   @db.VarChar(200)
  shortName   String?  @map("short_name") @db.VarChar(50)
  status      String   @default("active") @db.VarChar(20) // active / suspended
  legalRep    String?  @map("legal_rep") @db.VarChar(50) // 法人代表
  taxNo       String?  @map("tax_no") @db.VarChar(50) // 税号
  address     String?  @db.Text
  createdBy   String?  @map("created_by") @db.Uuid
  createdAt   DateTime @default(now()) @map("created_at") @db.Timestamptz(6)
  updatedAt   DateTime @updatedAt @map("updated_at") @db.Timestamptz(6)
  deletedAt   DateTime? @map("deleted_at") @db.Timestamptz(6)

  departments Department[]

  @@unique([code])
  @@index([status])
  @@map("companies")
}

model Department {
  id          String   @id @default(uuid()) @db.Uuid
  companyId   String   @map("company_id") @db.Uuid
  parentId    String?  @map("parent_id") @db.Uuid // 自引用，4 级层级上限（M0 V1.2 §二.2.1）
  code        String   @db.VarChar(50) // 部门编码（公司内唯一）
  name        String   @db.VarChar(200)
  leaderId    String?  @map("leader_id") @db.Uuid // 部门负责人（关联 employees）
  headcount   Int      @default(0) // 编制数
  order       Int      @default(0) // 同级排序
  status      String   @default("active") @db.VarChar(20) // active / suspended / merged
  createdBy   String?  @map("created_by") @db.Uuid
  createdAt   DateTime @default(now()) @map("created_at") @db.Timestamptz(6)
  updatedAt   DateTime @updatedAt @map("updated_at") @db.Timestamptz(6)
  deletedAt   DateTime? @map("deleted_at") @db.Timestamptz(6)

  company   Company      @relation(fields: [companyId], references: [id], onDelete: Restrict)
  parent    Department?  @relation("DepartmentHierarchy", fields: [parentId], references: [id], onDelete: SetNull)
  children  Department[] @relation("DepartmentHierarchy")
  employees Employee[]

  @@unique([companyId, code])
  @@index([companyId, parentId])
  @@index([status])
  @@map("departments")
}

// ==================== M1-A2 员工档案 ====================
// 8 类约 60 字段：基础 / 联系 / 教育 / 工作 / 合同 / 资质 / 银行卡 / 社保公积金
// 敏感字段（idCard / bankCard / phone）走 M0.5-3 crypto.service 加密
model Employee {
  id            String    @id @default(uuid()) @db.Uuid
  userId        String?   @unique @map("user_id") @db.Uuid // 1:1 关联 users（未建账号时为空）
  employeeNo    String    @unique @map("employee_no") @db.VarChar(30) // 工号
  name          String    @db.VarChar(50)
  // 基础信息
  gender        String?   @db.VarChar(10) // male / female
  birthDate     DateTime? @map("birth_date") @db.Date
  idCard        String?   @map("id_card") @db.Text // 加密存储（明文 → AES-256-GCM）
  nativePlace   String?   @map("native_place") @db.VarChar(200) // 籍贯
  ethnicity     String?   @db.VarChar(20) // 民族
  politicalStatus String? @map("political_status") @db.VarChar(50) // 政治面貌
  // 联系信息
  phone         String?   @map("phone") @db.Text // 加密存储
  email         String?   @db.VarChar(100)
  emergencyContactName String? @map("emergency_contact_name") @db.VarChar(50)
  emergencyContactPhone String? @map("emergency_contact_phone") @db.Text // 加密
  address       String?   @db.Text
  // 教育经历（单条最高学历，可扩展 JSON 数组）
  educationLevel String? @map("education_level") @db.VarChar(50) // 学历
  degree        String?   @db.VarChar(50) // 学位
  school        String?   @db.VarChar(200) // 院校
  major         String?   @db.VarChar(100) // 专业
  graduationDate DateTime? @map("graduation_date") @db.Date
  // 工作经历（JSON 数组 [{"employer":, "position":, "startDate":, "endDate":}])
  workHistory   Json?     @map("work_history")
  // 合同
  contractType  String?   @map("contract_type") @db.VarChar(50) // formal / intern / consultant / labor
  contractStart  DateTime? @map("contract_start") @db.Date
  contractEnd    DateTime? @map("contract_end") @db.Date
  // 资质证书（JSON 数组 [{"name":, "number":, "issueDate":, "expireDate":, "imageUrl":}])
  certificates   Json?
  // 银行卡
  bankName       String?  @map("bank_name") @db.VarChar(100)
  bankCard       String?  @map("bank_card") @db.Text // 加密
  // 社保公积金
  socialInsured   Boolean  @default(false) @map("social_insured")
  socialCity      String?  @map("social_city") @db.VarChar(50) // 西安 / 北京 / 四川
  socialBase      Decimal? @map("social_base") @db.Decimal(10, 2) // 缴费基数（加密 or 精度保留）
  housingFundCity String?  @map("housing_fund_city") @db.VarChar(50)
  housingFundRate Decimal? @map("housing_fund_rate") @db.Decimal(4, 2) // 5-12%
  housingFundBase Decimal? @map("housing_fund_base") @db.Decimal(10, 2)
  // 公司 / 部门
  companyId       String   @map("company_id") @db.Uuid
  departmentId    String   @map("department_id") @db.Uuid
  // 雇佣状态
  status          String   @default("probation") @db.VarChar(20) // probation / active / resigned
  hireDate        DateTime  @map("hire_date") @db.Date
  resignationDate DateTime? @map("resignation_date") @db.Date // 离职日期
  mustChangePassword Boolean @default(false) @map("must_change_password")
  // 备注
  remark          String?  @db.Text
  // 元数据
  createdBy       String?  @map("created_by") @db.Uuid
  createdAt       DateTime  @default(now()) @map("created_at") @db.Timestamptz(6)
  updatedAt       DateTime  @updatedAt @map("updated_at") @db.Timestamptz(6)
  deletedAt       DateTime? @map("deleted_at") @db.Timestamptz(6)

  company   Company   @relation(fields: [companyId], references: [id], onDelete: Restrict)
  department Department @relation(fields: [departmentId], references: [id], onDelete: Restrict)

  @@unique([companyId, departmentId, name, hireDate]) // 同公司同部门同名同入职日去重
  @@index([companyId, departmentId])
  @@index([status])
  @@index([contractEnd]) // 合同到期预警索引
  @@index([hireDate])
  @@index([deletedAt])
  @@map("employees")
}
